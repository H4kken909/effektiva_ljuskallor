"""A11 automatiserad: prompten med samma kommandon som för hand, Ctrl+C och Ctrl+Z simulerade."""
import _thread
import builtins
import os
import sys
import threading

sys.path.insert(0, os.getcwd())
os.environ.setdefault("MPLBACKEND", "Agg")

from spectrometer import main as programmet  # noqa: E402

KORNING_1 = [
    "source file tests/data/bilder/rand_x12.png",
    "help",
    "roi 100 200 400",
    "channel purple",
    "foo",
    "show",
    "frames 200000",
    "step",
    "frames 1",
    "quit",
]


def kor(rader):
    """Kör prompten en gång med raderna som om de skrevs; KeyboardInterrupt och EOFError som tangenterna."""
    flode = iter(rader)

    def tangentbord(prompt=""):
        """Skriver prompten och raden som i terminalen, eller tangenten Ctrl+Z eller Ctrl+C."""
        sys.stdout.write(prompt)
        sys.stdout.flush()
        rad = next(flode, EOFError)
        if rad is EOFError:
            print("^Z")
            raise EOFError
        if rad is KeyboardInterrupt:
            print("^C")
            raise KeyboardInterrupt
        print(rad)
        if rad == "step":
            threading.Timer(0.5, _thread.interrupt_main).start()
        return rad

    builtins.input = tangentbord
    sys.argv = ["spectrometer.main"]
    print("PS> python -m spectrometer.main")
    sys.stdout.flush()
    programmet.main()
    sys.stdout.flush()


kor(KORNING_1)
kor([])
kor([KeyboardInterrupt])
